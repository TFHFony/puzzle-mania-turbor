
;;FILM
; ================================================================================================ the FONY film: SCREEN 5, 256 x 144, span updates, PCM sound, Z80 mode
fl_sp   equ GV + 120                    ; the stream pointer while the housekeeping runs
fl_bank equ GV + 254                    ; the bank of the stream (page 1 window)
fl_in_bank  equ GV + 122                ; input of film_run: the first stream bank, the first audio bank, R#23, the palette (32 bytes)
fl_in_abank equ GV + 124
fl_in_r23   equ GV + 126
fl_gb       equ GV + 127                ; (the groups of region B of the frame)
fl_in_pal   equ GV + 128
fl_half     equ GV + 132
fl_skip     equ GV + 131                ; 1: the film was stopped with SPACE

; film_play: the FONY intro film
film_play:
        ld hl,FILM0
        ld (fl_in_bank),hl
        ld hl,FILMA0
        ld (fl_in_abank),hl
        ld hl,film_pal
        ld (fl_in_pal),hl
        ld a,222                        ; (the film at screen line 34)
        ld (fl_in_r23),a
        jp film_run

; tv_play: the title video in the letters + the jigsaw outro, then XXXL
tv_play:
        ld hl,TV0
        ld (fl_in_bank),hl
        ld hl,TVA0
        ld (fl_in_abank),hl
        ld hl,tv_pal
        ld (fl_in_pal),hl
        ld a,246                        ; (the picture of 192 lines at screen line 10)
        ld (fl_in_r23),a
        call film_run
        ld a,(fl_skip)
        or a
        jp nz,choice_enter              ; SPACE: straight to the menu
        call xxxl_show
        jp choice_enter

; att_after: the attract mode has played its clip: the title (the video) again, then the choice screen
att_after:
        xor a
        ld (g_att),a
        ld (g_aph),a
        ld a,200
        ld (g_shn),a
        call att_unmute
        jp tv_play

; film_run: plays a span stream (until its end or SPACE), then back to the R800 and SCREEN 11; the music pointers are kept
film_run:
        ld hl,(au_bank)
        push hl
        ld hl,(au_ptr)
        push hl
        ld hl,(au_start)
        push hl
        ld hl,(au_end)
        push hl
        call wait_idle
        di
        xor a                           ; the display off until the picture memory is clear (no junk of the old screen)
        out (0x99),a
        ld a,0x81
        out (0x99),a
        ld a,0x80                       ; Z80 mode
        call chgcpu
        di
        ld hl,f5regs                    ; SCREEN 5
        ld b,f5end - f5regs
        ld c,0x80
.vr:    ld a,(hl)
        out (0x99),a
        ld a,c
        out (0x99),a
        inc hl
        inc c
        djnz .vr
        ld a,(g_r9)
        out (0x99),a
        ld a,0x89
        out (0x99),a
        xor a                           ; R#25 = 0, R#18 = 0, R#23
        out (0x99),a
        ld a,0x99
        out (0x99),a
        xor a
        out (0x99),a
        ld a,0x92
        out (0x99),a
        ld a,(fl_in_r23)
        out (0x99),a
        ld a,0x97
        out (0x99),a
        xor a                           ; the palette
        out (0x99),a
        ld a,0x90
        out (0x99),a
        ld hl,(fl_in_pal)
        ld bc,0x209A
        otir
        ld b,2                          ; clear the VRAM (32 KB: page 0): R#14 = 0 and 1
.cp:    ld a,b
        dec a
        xor 1
        out (0x99),a
        ld a,0x8E
        out (0x99),a
        xor a
        out (0x99),a
        ld a,0x40
        out (0x99),a
        ld hl,0x4000
.cl:    xor a
        out (0x98),a
        dec hl
        ld a,h
        or l
        jr nz,.cl
        djnz .cp
        ld a,0x40                       ; the display on (R#1)
        out (0x99),a
        ld a,0x81
        out (0x99),a
        ld hl,(fl_in_bank)
        ld (fl_bank),hl
        ld a,l
        ld (0x6000),a
        ld a,h
        ld (0x6001),a
        ld hl,(fl_in_abank)
        ld (au_bank),hl
        ld a,l
        ld (0x7000),a
        ld a,h
        ld (0x7001),a
        ld de,0x8000
        ld hl,0x4000
        ld c,0x98
        jp film_frame
f5regs: db 0x06, 0x00, 0x1F, 0xFF, 0x03, 0x36, 0x07, 0x00, 0x0A
f5end:

        INCLUDE "filmdata.inc"
        INCLUDE "filmplay.inc"

film_end:                               ; the end of the film
        xor a
        jr film_out
film_exit:                              ; SPACE
        ld a,1
film_out:
        ld (fl_skip),a
        ld a,0x82
        call chgcpu
        di
        call screen11_init
        xor a                           ; R#23 = 0, R#14 = 0
        out (0x99),a
        ld a,0x97
        out (0x99),a
        xor a
        out (0x99),a
        ld a,0x8E
        out (0x99),a
        in a,(0xE6)
        ld (tgt),a
        pop hl
        ld (au_end),hl
        pop hl
        ld (au_start),hl
        pop hl
        ld (au_ptr),hl
        pop hl
        ld (au_bank),hl
        ld a,l
        ld (0x7000),a
        ld a,h
        ld (0x7001),a
        ret
; xxxl_show: the letters XXXL with the photo of the Oops clip and the dreun of the FONY film (until it is over or a key)
xxxl_show:
        xor a                           ; the display off while the picture is drawn: it appears at once (R#1 = 0)
        out (0x99),a
        ld a,0x81
        out (0x99),a
        ld e,0                          ; the picture: 4 banks of 16 KB straight to the VRAM (R#14 = part), no queue: well under a second
.xp:    ld a,e
        out (0x99),a
        ld a,0x8E
        out (0x99),a
        xor a
        out (0x99),a
        ld a,0x40
        out (0x99),a
        ld hl,XXXL0
        ld d,0
        add hl,de
        ld a,l
        ld (0x6000),a
        ld a,h
        ld (0x6001),a
        ld hl,0x4000
        ld c,0x98
        ld d,64
.xl:    ld b,0
        otir
        dec d
        jr nz,.xl
        inc e
        ld a,e
        cp 4
        jr nz,.xp
        xor a                           ; R#14 = 0
        out (0x99),a
        ld a,0x8E
        out (0x99),a
        ld a,0x40                       ; the display on
        out (0x99),a
        ld a,0x81
        out (0x99),a
        ld hl,(au_bank)
        push hl
        ld hl,(au_ptr)
        push hl
        ld hl,(au_start)
        push hl
        ld hl,(au_end)
        push hl
        ld hl,BOOMB                     ; the dreun: the end of the film audio
        ld de,FILMA0 + FILMAN
        call au_play
        ld hl,0x8000 + BOOMO
        ld (au_ptr),hl
        ld a,2                          ; (two waits: the sample is longer than 32767 counts)
        ld (fl_half),a
        ld hl,(smp)
        ld de,BOOMLEN / 2
        add hl,de
        ld (g_tm2),hl
.rel:   call tick                       ; (the key that ended the video must be up)
        ld c,8
        call rdrow
        and 1
        jr nz,.rel
.w:     call tick
        ld c,8
        call rdrow
        and 1
        jr nz,.done
        ld hl,(smp)
        ld de,(g_tm2)
        or a
        sbc hl,de
        bit 7,h
        jr nz,.w
        ld a,(fl_half)
        dec a
        ld (fl_half),a
        jr z,.done
        ld hl,(g_tm2)
        ld de,BOOMLEN / 2
        add hl,de
        ld (g_tm2),hl
        jr .w
.done:  pop hl
        ld (au_end),hl
        pop hl
        ld (au_start),hl
        pop hl
        ld (au_ptr),hl
        pop hl
        ld (au_bank),hl
        ld a,l
        ld (0x7000),a
        ld a,h
        ld (0x7001),a
.r2:    call tick
        ld c,8
        call rdrow
        and 1
        jr nz,.r2
        ret
;;ENDFILM
