
;;CLIP
; ================================================================================================ the reward clip: SCREEN 2 with the MSX2 palette of the photo, PCM sound, Z80 mode
cl_bank equ GV + 110                    ; the bank of the next frame
cl_cnt  equ GV + 112                    ; frames left in the clip
cl_b0   equ GV + 114                    ; the first bank / the number of frames of the clip
cl_n0   equ GV + 116
cl_pb   equ GV + 118                    ; the palette stream: the bank and the address in the window
cl_pp   equ GV + 250
cl_pb0  equ GV + 252
        INCLUDE "clipdata.inc"          ; clip_tab (bank, frames) and clip_pal (32 bytes per photo)

; clip_start: A = the photo; the clip plays until SPACE, then the photo menu
clip_start:
        ld (g_tmp),a
        ld l,a
        ld h,0
        add hl,hl
        add hl,hl
        ld de,clip_tab
        add hl,de
        ld e,(hl)
        inc hl
        ld d,(hl)
        inc hl
        ld (cl_b0),de
        ld (cl_bank),de
        ld e,(hl)
        inc hl
        ld d,(hl)
        ld (cl_n0),de
        ld (cl_cnt),de
        ld a,(g_tmp)                    ; the palette stream of the clip
        ld l,a
        ld h,0
        add hl,hl
        ld de,clip_ptab
        add hl,de
        ld e,(hl)
        inc hl
        ld d,(hl)
        ld (cl_pb),de
        ld (cl_pb0),de
        ld hl,0x4000
        ld (cl_pp),hl
.rel:   call tick                       ; SPACE (the key that started the clip) must be up
        ld c,8
        call rdrow
        and 1
        jr nz,.rel
        call wait_idle
        di
        ld a,0x80                       ; Z80 mode (the VDP is faster for the CPU there)
        call chgcpu
        di
        ld hl,s2regs                    ; SCREEN 2
        ld b,s2end - s2regs
        ld c,0x80
.vr:    ld a,(hl)
        out (0x99),a
        ld a,c
        out (0x99),a
        inc hl
        inc c
        djnz .vr
        xor a                           ; R#14 = 0 (the last job of SCREEN 11 left A16 set), R#25 = 0 (no YJK)
        out (0x99),a
        ld a,0x8E
        out (0x99),a
        xor a
        out (0x99),a
        ld a,0x99
        out (0x99),a
        ld a,(g_r9)
        and 2
        out (0x99),a
        ld a,0x89
        out (0x99),a
        xor a                           ; the palette of the photo
        out (0x99),a
        ld a,0x90
        out (0x99),a
        ld a,(g_tmp)                    ; the first 32 bytes of the palette stream of the clip = the palette of its first frame
        add a,a
        ld l,a
        ld h,0
        ld de,clip_ptab
        add hl,de
        ld e,(hl)
        inc hl
        ld d,(hl)
        ld a,e
        ld (0x6000),a
        ld a,d
        ld (0x6001),a
        ld hl,0x4000
        ld bc,0x209A
        otir
        xor a                           ; the name table 0..255 three times
        out (0x99),a
        ld a,0x58
        out (0x99),a
        ld c,3
.nt:    xor a
.n2:    out (0x98),a
        inc a
        jr nz,.n2
        dec c
        jr nz,.nt
        ld a,0xD0                       ; (the sprite attribute table follows the name table: the end of the sprite list; the old SCREEN 11 picture lies there and would show as sprites)
        out (0x98),a
        ld de,(au_ptr)
        jp clip_frame
s2regs: db 0x02, 0x00, 0x06, 0xFF, 0x03, 0x36, 0x07, 0x00, 0x0A, 0x00, 0x00, 0x00   ; (R#1 = 0: the display stays off until the first frame is complete; R#10 = 0; R#11 = 0: the sprite attribute table at 1B00h, not at 9B00h as the cursor of the menus left it)
s2end:

        INCLUDE "clipplay.inc"

clip_wrap:                              ; the end of the clip: the first frame again
        ld hl,(cl_b0)
        ld (cl_bank),hl
        ld hl,(cl_n0)
        ld (cl_cnt),hl
        ld hl,(cl_pb0)
        ld (cl_pb),hl
        ld hl,0x4000
        ld (cl_pp),hl
        jp clip_after

clip_exit:                              ; SPACE: back to the R800 and SCREEN 11
        ld (au_ptr),de
        ld a,0x82
        call chgcpu
        di
        call screen11_init
        in a,(0xE6)
        ld (tgt),a
        ld a,(g_after)
        or a
        jp z,menu_enter
        jp choice_enter

; chgcpu: the BIOS call CHGCPU (A = mode) on a stack of its own: it needs about 90 bytes, the stack of the game is shorter (the queue buffer is idle: wait_idle)
chgcpu:
        ld (0xC0B6),sp
        ld sp,0xC0B0
        call 0x0180
        ld sp,(0xC0B6)
        ret

; the VDP registers and the palette of SCREEN 11 (the same as the start-up code)
screen11_init:
        ld hl,s11regs
        ld b,9
        ld c,0x80
.vr:    ld a,(hl)
        out (0x99),a
        ld a,c
        out (0x99),a
        inc hl
        inc c
        djnz .vr
        ld a,(g_r9)
        out (0x99),a
        ld a,0x89
        out (0x99),a
        ld hl,s11regs2
        ld b,5
.v2:    ld a,(hl)
        out (0x99),a
        inc hl
        ld a,(hl)
        out (0x99),a
        inc hl
        djnz .v2
        xor a
        out (0x99),a
        ld a,0x90
        out (0x99),a
        ld hl,s11pal
        ld bc,0x209A
        otir
        ret
s11regs: db 0x0E, 0x40, 0x1F, 0, 0, 0, 0, 0x01, 0x0A
s11regs2: db 0x00, 0x92, 0x00, 0x97, 0x18, 0x99, 0x00, 0x9A, 0x00, 0x9B
s11pal: db 0x00, 0x00, 0x00, 0x00, 0x11, 0x05, 0x33, 0x06, 0x27, 0x02, 0x37, 0x03, 0x25, 0x02, 0x27, 0x06
        db 0x72, 0x02, 0x73, 0x03, 0x62, 0x06, 0x63, 0x06, 0x14, 0x01, 0x65, 0x02, 0x55, 0x05, 0x77, 0x07
;;ENDCLIP
